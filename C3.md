' ============================================================
' 直接處理 JScriptTypeInfo，不需要先轉 Dictionary
' ============================================================

Function ToCompactJson(val As Variant, Optional filterKeys As Variant) As String

    Dim i As Long
    Dim key As Variant
    
    Dim hasFilter As Boolean
    hasFilter = Not IsMissing(filterKeys)

    ' ── 情況①：Null / Empty ───────────────────────────────
    If IsNull(val) Or IsEmpty(val) Then
        ToCompactJson = ""
        Exit Function
    End If

    ' ── 情況②：Array ──────────────────────────────────────
    If IsArray(val) Then
        Dim arrParts() As String
        Dim arrCount As Long
        arrCount = 0
        ReDim arrParts(LBound(val) To UBound(val))
        
        For i = LBound(val) To UBound(val)
            Dim arrItem As String
            arrItem = ToCompactJson(val(i))
            If arrItem <> "" Then
                arrParts(arrCount) = arrItem
                arrCount = arrCount + 1
            End If
        Next i
        
        If arrCount = 0 Then
            ToCompactJson = "[]"
        Else
            ReDim Preserve arrParts(0 To arrCount - 1)
            ToCompactJson = "[" & Join(arrParts, ",") & "]"
        End If

    ' ── 情況③：Dictionary ─────────────────────────────────
    ElseIf TypeName(val) = "Dictionary" Then
        Dim objParts() As String
        Dim objCount As Long
        objCount = 0
        ReDim objParts(0 To val.Count - 1)
        
        For Each key In val.Keys
            If hasFilter Then
                Dim keyFound As Boolean
                keyFound = False
                Dim j As Long
                For j = LBound(filterKeys) To UBound(filterKeys)
                    If StrComp(CStr(key), CStr(filterKeys(j)), vbTextCompare) = 0 Then
                        keyFound = True
                        Exit For
                    End If
                Next j
                If Not keyFound Then GoTo SkipDictKey
            End If
            
            Dim dictVal As String
            dictVal = ToCompactJson(val(key))
            If dictVal <> "" Then
                objParts(objCount) = """" & CStr(key) & """:" & dictVal
                objCount = objCount + 1
            End If
SkipDictKey:
        Next key
        
        If objCount = 0 Then
            ToCompactJson = "{}"
        Else
            ReDim Preserve objParts(0 To objCount - 1)
            ToCompactJson = "{" & Join(objParts, ",") & "}"
        End If

    ' ── 情況④：JScriptTypeInfo ────────────────────────────
    ' XMLHTTP で取得した JSON を ScriptControl で eval() すると
    ' このタイプになる
    ElseIf TypeName(val) = "JScriptTypeInfo" Then
    
        ' ScriptControl を使って JS の Object.keys() でキー一覧を取得
        Dim sc As Object
        Set sc = CreateObject("ScriptControl")
        sc.Language = "JScript"
        
        ' val が配列かどうかを JS 側で判定
        ' Array.isArray() は JS のネイティブ関数
        Dim isJsArray As Boolean
        isJsArray = sc.Eval("Array.isArray(arguments[0])"  ) ' ← これは動かない
        
        ' ★ JScriptTypeInfo を JS に渡す方法は ExecuteStatement が確実
        ' まず JSON.stringify で文字列化して判定する
        Dim jsonStr As String
        jsonStr = sc.Eval("JSON.stringify(arguments[0])")
        ' ↑ arguments[0] は使えないので下記の方法を使う
        
        ' ── 正しいやり方：一時変数に格納してから操作 ──
        ' sc.ExecuteStatement で JS 変数に代入はできないが
        ' CallByName でプロパティを読むことはできる
        
        ' まず JSON 文字列に変換（val 自体の JSON.stringify）
        ' JScriptTypeInfo には独自の .toString() がないので
        ' sc を経由して文字列化する
        
        ' val の先頭文字で配列か Object か判定
        Dim firstChar As String
        ' CallByName でアクセスできないので length プロパティで判断
        On Error Resume Next
        Dim testLen As Variant
        testLen = CallByName(val, "length", VbGet)
        On Error GoTo 0
        
        If Not IsEmpty(testLen) And Not IsNull(testLen) Then
            ' ── JS Array として処理 ──────────────────
            Dim jsArrParts() As String
            Dim jsArrCount As Long
            jsArrCount = 0
            ReDim jsArrParts(0 To CLng(testLen) - 1)
            
            For i = 0 To CLng(testLen) - 1
                Dim jsArrItem As String
                ' 配列の各要素は CallByName(val, インデックス文字列, VbGet) で取得
                Dim elem As Variant
                On Error Resume Next
                elem = CallByName(val, CStr(i), VbGet)
                On Error GoTo 0
                
                jsArrItem = ToCompactJson(elem)
                If jsArrItem <> "" Then
                    jsArrParts(jsArrCount) = jsArrItem
                    jsArrCount = jsArrCount + 1
                End If
            Next i
            
            If jsArrCount = 0 Then
                ToCompactJson = "[]"
            Else
                ReDim Preserve jsArrParts(0 To jsArrCount - 1)
                ToCompactJson = "[" & Join(jsArrParts, ",") & "]"
            End If
            
        Else
            ' ── JS Object として処理 ─────────────────
            ' Object.keys() でキー一覧を取得する
            ' ExecuteStatement で __vbaKeys 変数に格納
            sc.ExecuteStatement "var __vbaKeys = Object.keys(" & _
                                 "eval('(' + JSON.stringify(arguments[0]) + ')')" & _
                                 ");"
            ' ↑ arguments[0] は ExecuteStatement では使えない
            
            ' ★ 現実的な解決策：
            '   CallByName でアクセスできるのはプロパティ名がわかっている時だけ
            '   キー一覧を取得するには JSON.stringify → Split が最も確実
            
            ' JSON.stringify を直接 val に対して呼ぶ
            ' JScriptTypeInfo は sc.Eval に直接渡せないが
            ' sc.Run でラッパー関数経由なら渡せる
            sc.ExecuteStatement "function __getKeys(o){ return Object.keys(o).join(','); }"
            Dim keysStr As String
            keysStr = sc.Run("__getKeys", val)
            ' keysStr = "name,age,address" のような文字列
            
            If keysStr = "" Then
                ToCompactJson = "{}"
                Exit Function
            End If
            
            Dim jsKeys() As String
            jsKeys = Split(keysStr, ",")
            
            Dim jsParts() As String
            Dim jsCount As Long
            jsCount = 0
            ReDim jsParts(0 To UBound(jsKeys))
            
            For i = 0 To UBound(jsKeys)
                Dim jsKey As String
                jsKey = jsKeys(i)
                
                ' filterKeys チェック
                If hasFilter Then
                    Dim jsKeyFound As Boolean
                    jsKeyFound = False
                    Dim fj As Long
                    For fj = LBound(filterKeys) To UBound(filterKeys)
                        If StrComp(jsKey, CStr(filterKeys(fj)), vbTextCompare) = 0 Then
                            jsKeyFound = True
                            Exit For
                        End If
                    Next fj
                    If Not jsKeyFound Then GoTo SkipJsKey
                End If
                
                ' CallByName でプロパティ値を取得
                Dim jsVal As Variant
                On Error Resume Next
                jsVal = CallByName(val, jsKey, VbGet)
                On Error GoTo 0
                
                Dim jsValStr As String
                jsValStr = ToCompactJson(jsVal)
                
                If jsValStr <> "" Then
                    jsParts(jsCount) = """" & jsKey & """:" & jsValStr
                    jsCount = jsCount + 1
                End If
SkipJsKey:
            Next i
            
            If jsCount = 0 Then
                ToCompactJson = "{}"
            Else
                ReDim Preserve jsParts(0 To jsCount - 1)
                ToCompactJson = "{" & Join(jsParts, ",") & "}"
            End If
        End If
        
        Set sc = Nothing

    ' ── 情況⑤：Boolean ────────────────────────────────────
    ElseIf TypeName(val) = "Boolean" Then
        ToCompactJson = IIf(val, "true", "false")

    ' ── 情況⑥：數字 ──────────────────────────────────────
    ElseIf IsNumeric(val) Then
        ToCompactJson = CStr(val)

    ' ── 情況⑦：字串 ──────────────────────────────────────
    Else
        Dim s As String
        s = CStr(val)
        s = Replace(s, "\", "\\")
        s = Replace(s, """", "\""")
        s = Replace(s, Chr(10), "\n")
        s = Replace(s, Chr(13), "\r")
        s = Replace(s, Chr(9), "\t")
        ToCompactJson = """" & s & """"
    End If

End Function


' ── 実際の使用例 ──────────────────────────────────────────
Sub TestXmlHttp()
    Dim http As Object
    Set http = CreateObject("MSXML2.XMLHTTP")
    http.Open "GET", "https://api.example.com/data", False
    http.Send
    
    ' responseText を ScriptControl で eval
    Dim sc As Object
    Set sc = CreateObject("ScriptControl")
    sc.Language = "JScript"
    
    Dim parsed As Object
    Set parsed = sc.Eval("(" & http.responseText & ")")
    ' parsed は JScriptTypeInfo
    
    ' そのまま渡すだけ
    Debug.Print ToCompactJson(parsed)
    Debug.Print ToCompactJson(parsed, Array("name", "age"))
End Sub
