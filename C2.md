' ============================================================
' 步驟一：用 ScriptControl 把 JSON 字串解析成 JScriptTypeInfo
'         再遞迴轉成 VBA Dictionary / Array
' ============================================================

' 需要引用：
'   - Microsoft Scripting Runtime（Dictionary 用）
'   - Microsoft Script Control 1.0（ScriptControl 用）

' ── 入口函數：傳入 JSON 字串，直接拿到壓縮 JSON ──────────
Function JsonStringToCompact(jsonStr As String) As String
    Dim sc As New ScriptControl
    sc.Language = "JScript"
    
    ' eval() 把 JSON 字串解析成 JScriptTypeInfo 物件
    Dim parsed As Object
    Set parsed = sc.Eval("(" & jsonStr & ")")
    
    ' 轉成 Dictionary 再丟給 ToCompactJson
    Dim dict As Object
    Set dict = JsObjectToDict(parsed, sc)
    
    JsonStringToCompact = ToCompactJson(dict)
End Function

' ── 核心轉換：JScriptTypeInfo → Dictionary ────────────────
' sc 要一直傳下去，因為遞迴時還需要用它執行 JS 來取得 keys
Function JsObjectToDict(jsObj As Object, sc As Object) As Object

    ' 用 JS 的 Object.keys() 取得所有 key 名稱，用逗號串起來
    ' 例如 {"name":"Azusa","age":30} → "name,age"
    Dim keysStr As String
    keysStr = sc.Eval("Object.keys(" & sc.Eval("JSON.stringify") & ").toString()")
    
    ' ↑ 上面那行太複雜，改用下面這個更穩的寫法：
    ' 把物件存到 JS 的暫存變數 __tmp，再對它呼叫 Object.keys()
    sc.ExecuteStatement "var __tmp = " & sc.Eval("JSON.stringify(arguments[0])") 
    
    ' 實際上最簡單穩定的做法：直接讓 JS 幫我們序列化再解析 key
    Dim result As New Scripting.Dictionary
    
    ' 讓 JScript 把物件的 keys 輸出成 JSON 陣列字串
    ' 例如 → ["name","age","address"]
    sc.ExecuteStatement "var __obj = " & JsonStringOf(jsObj, sc)
    keysStr = sc.Eval("JSON.stringify(Object.keys(__obj))")
    
    ' keysStr 現在是 ["name","age"] 這樣的字串
    ' 用 Split 切開（去掉頭尾的 [ ] " ）
    keysStr = Mid(keysStr, 2, Len(keysStr) - 2)  ' 去掉 [ 和 ]
    
    Dim keys() As String
    keys = Split(keysStr, ",")
    
    Dim i As Long
    For i = 0 To UBound(keys)
        ' 去掉每個 key 前後的雙引號
        Dim k As String
        k = Mid(keys(i), 2, Len(keys(i)) - 2)
        
        ' 用 CallByName 取得這個 key 的值
        Dim childVal As Variant
        childVal = CallByName(jsObj, k, VbGet)
        
        ' 判斷子值的型別，決定怎麼處理
        Select Case TypeName(childVal)
            Case "JScriptTypeInfo"
                ' 子物件 → 遞迴轉成 Dictionary
                result(k) = JsObjectToDict(childVal, sc)
            Case "Null", "Empty"
                ' Null 跳過（不加進 Dictionary）
            Case Else
                ' 數字、字串、Boolean 直接存
                result(k) = childVal
        End Select
    Next i
    
    Set JsObjectToDict = result
End Function

' ── 輔助：把 JScriptTypeInfo 序列化回 JSON 字串（給 JS 用）
Function JsonStringOf(jsObj As Object, sc As Object) As String
    sc.ExecuteStatement "var __serialize = function(o){ return JSON.stringify(o); }"
    JsonStringOf = sc.Run("__serialize", jsObj)
End Function
